5. Web Audio API Precision Synthesizer

The following script delivers a specialized 869 Hz synthesis engine equipped with low-frequency splanchnic grounding and smooth fading:

/**
 * Rife869Synthesizer - Precision Web Audio Engine for 869 Hz Resonance
 * Tailored for hepatoprotective and systemic antiviral entrainment.
 */
class Rife869Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.hepaticGround = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioContextClass();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 40) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Primary 869 Hz Oscillator
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(869, t);

    // 434.5 Hz Hepatic Portal Sub-Harmonic
    this.hepaticGround = this.audioCtx.createOscillator();
    this.hepaticGround.type = 'sine';
    this.hepaticGround.frequency.setValueAtTime(434.5, t);

    const groundGain = this.audioCtx.createGain();
    groundGain.gain.setValueAtTime(0.25, t);

    this.carrier.connect(this.masterGain);
    this.hepaticGround.connect(groundGain);
    groundGain.connect(this.masterGain);

    // Smooth envelope attack
    this.masterGain.gain.exponentialRampToValueAtTime(0.28, t + 3);

    this.carrier.start(t);
    this.hepaticGround.start(t);
    this.isActive = true;

    // Auto-stop timer
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.28, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.hepaticGround.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1.5);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.hepaticGround) this.hepaticGround.stop();
      this.isActive = false;
    }, 1500);
  }
}
