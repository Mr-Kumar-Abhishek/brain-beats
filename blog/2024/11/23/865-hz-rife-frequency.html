5. Web Audio API Generator

Below is an implementation of an 865 Hz acoustic synthesis engine utilizing the Web Audio API. It features dual-oscillator phase alignment, gentle binaural beat modulation, and smooth envelope ramps.

/**
 * Rife865Synthesizer - Web Audio API Engine for 865 Hz Resonance Therapy
 * Provides dual-channel phase-coherent synthesis with 4 Hz theta modulation.
 */
class Rife865Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.primaryOsc = null;
    this.detuneOsc = null;
    this.gainNode = null;
    this.isPlaying = false;
  }

  init() {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioContextClass();

    this.gainNode = this.audioCtx.createGain();
    this.gainNode.gain.setValueAtTime(0.001, this.audioCtx.currentTime);
    this.gainNode.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isPlaying) return;

    const now = this.audioCtx.currentTime;

    // Primary Fundamental Oscillator (865 Hz)
    this.primaryOsc = this.audioCtx.createOscillator();
    this.primaryOsc.type = 'sine';
    this.primaryOsc.frequency.setValueAtTime(865, now);

    // Subtle Detuned Binaural Oscillator (869 Hz for 4 Hz Theta entrainment)
    this.detuneOsc = this.audioCtx.createOscillator();
    this.detuneOsc.type = 'sine';
    this.detuneOsc.frequency.setValueAtTime(869, now);

    // Connect to gain envelope
    this.primaryOsc.connect(this.gainNode);
    this.detuneOsc.connect(this.gainNode);

    // Fade-in envelope
    this.gainNode.gain.linearRampToValueAtTime(0.3, now + 3);

    this.primaryOsc.start(now);
    this.detuneOsc.start(now);
    this.isPlaying = true;

    // Auto ramp-down after duration
    const stopTime = now + (durationMinutes * 60);
    this.gainNode.gain.setValueAtTime(0.3, stopTime - 3);
    this.gainNode.gain.linearRampToValueAtTime(0.001, stopTime);
    this.primaryOsc.stop(stopTime);
    this.detuneOsc.stop(stopTime);

    this.primaryOsc.onended = () => {
      this.isPlaying = false;
    };
  }

  stop() {
    if (!this.isPlaying || !this.audioCtx) return;
    const now = this.audioCtx.currentTime;
    this.gainNode.gain.linearRampToValueAtTime(0.001, now + 1.5);
    setTimeout(() => {
      if (this.primaryOsc) this.primaryOsc.stop();
      if (this.detuneOsc) this.detuneOsc.stop();
      this.isPlaying = false;
    }, 1500);
  }
}
