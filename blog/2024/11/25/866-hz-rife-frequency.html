5. Web Audio API Master Synthesizer

The following implementation provides a complete, standalone 866 Hz sound generator equipped with dynamic panning, soft-clip wave shaping, and gentle ramp transitions:

/**
 * Rife866MasterEngine - Precision Web Audio API Synthesizer for 866 Hz
 * Features harmonic wave shaping, binaural spatialization, and safe gain ramps.
 */
class Rife866MasterEngine {
  constructor() {
    this.audioCtx = null;
    this.osc1 = null;
    this.osc2 = null;
    this.waveshaper = null;
    this.gain = null;
    this.isRunning = false;
  }

  init() {
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioContext();

    // Soft-clip saturation curve to generate subtle, therapeutic odd harmonics
    this.waveshaper = this.audioCtx.createWaveShaper();
    this.waveshaper.curve = this._makeDistortionCurve(5);
    this.waveshaper.oversample = '4x';

    this.gain = this.audioCtx.createGain();
    this.gain.gain.setValueAtTime(0.001, this.audioCtx.currentTime);

    this.waveshaper.connect(this.gain);
    this.gain.connect(this.audioCtx.destination);
  }

  _makeDistortionCurve(k = 5) {
    const n_samples = 44100;
    const curve = new Float32Array(n_samples);
    const deg = Math.PI / 180;
    for (let i = 0; i < n_samples; ++i) {
      const x = (i * 2) / n_samples - 1;
      curve[i] = ((3 + k) * x * 20 * deg) / (Math.PI + k * Math.abs(x));
    }
    return curve;
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isRunning) return;

    const t = this.audioCtx.currentTime;

    // Primary 866 Hz Oscillator
    this.osc1 = this.audioCtx.createOscillator();
    this.osc1.type = 'sine';
    this.osc1.frequency.setValueAtTime(866, t);

    // Binaural Secondary Oscillator (872 Hz = 6 Hz Theta entrainment)
    this.osc2 = this.audioCtx.createOscillator();
    this.osc2.type = 'sine';
    this.osc2.frequency.setValueAtTime(872, t);

    this.osc1.connect(this.waveshaper);
    this.osc2.connect(this.waveshaper);

    // Smooth envelope ramp
    this.gain.gain.exponentialRampToValueAtTime(0.25, t + 3);

    this.osc1.start(t);
    this.osc2.start(t);
    this.isRunning = true;

    // Stop timer
    const endTime = t + (durationMinutes * 60);
    this.gain.gain.setValueAtTime(0.25, endTime - 3);
    this.gain.gain.exponentialRampToValueAtTime(0.001, endTime);
    this.osc1.stop(endTime);
    this.osc2.stop(endTime);

    this.osc1.onended = () => {
      this.isRunning = false;
    };
  }

  stop() {
    if (!this.isRunning || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.gain.gain.linearRampToValueAtTime(0.001, t + 1.5);
    setTimeout(() => {
      if (this.osc1) this.osc1.stop();
      if (this.osc2) this.osc2.stop();
      this.isRunning = false;
    }, 1500);
  }
}
