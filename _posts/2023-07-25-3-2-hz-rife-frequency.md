### 1. Microbiological & Pathological Profile

- **Biliary Ductal Peristalsis & Fluid Resistance**: The common bile duct exhibits coordinated myogenic rhythmic contractions that work in tandem with duodenal cholecystokinin (CCK) release to propel concentrated bile into the intestinal lumen. Biliary stasis and chronic low-grade inflammation cause fibrotic ductal wall thickening and increased hydraulic resistance, creating a favorable niche for Gram-negative bacterial colonization and stone nidus aggregation.
- **Pharmacotherapy & openFDA Clearances**: Medical management of cholestatic sludging and lithiasis relies on chronic hydrophilic bile acid therapy with Ursodiol (Actigall, NDA 019594). Acute infectious complications require parenteral antimicrobial agents with high biliary excretion rates, such as Piperacillin-tazobactam (Zosyn, NDA 050684) and Ciprofloxacin (Cipro, NDA 019537). Mechanical ductal decompression requires surgical or endoscopic stenting. Sound waves do not possess antimicrobial potency or the mechanical torque needed to clear impacted stones.

### 5. Web Audio API Master Synthesizer

```javascript
/**
 * Rife3_2Synthesizer - Web Audio API Synthesis Engine for 3.2 Hz
 * 3.2 Hz is infrasonic, so it is rendered via amplitude modulation of a 200 Hz audible carrier.
 */
class Rife3_2Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.lfo = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Audible carrier wave (200 Hz pure sine)
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(200, t);

    // Infrasonic LFO modulator at 3.2 Hz
    this.lfo = this.audioCtx.createOscillator();
    this.lfo.type = 'sine';
    this.lfo.frequency.setValueAtTime(3.2, t);

    // Modulator gain depth and baseline bias
    const depth = this.audioCtx.createGain();
    depth.gain.setValueAtTime(0.5, t);
    const amp = this.audioCtx.createGain();
    amp.gain.setValueAtTime(0.5, t);

    this.lfo.connect(depth);
    depth.connect(amp.gain);
    this.carrier.connect(amp);
    amp.connect(this.masterGain);

    // Smooth linear attack envelope (3 seconds)
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrier.start(t);
    this.lfo.start(t);
    this.isActive = true;

    // Automatic session shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.lfo.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.lfo) this.lfo.stop();
      this.isActive = false;
    }, 1000);
  }
}
```
