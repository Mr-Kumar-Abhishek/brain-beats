### 1. Microbiological & Pathological Profile

- **Biliary System Pathophysiology & Cholestasis**: The biliary system coordinates the transport of bile acids, cholesterol, and bilirubin from hepatocyte canaliculi through intrahepatic and extrahepatic bile ducts into the duodenum. Pathological cholestasis, choledochal micro-stones, and biliary dyskinesia cause retrograde intrabiliary hypertension, ductal epithelial injury, and pruritus. Untreated biliary stasis increases the risk of acute ascending bacterial cholangitis (Charcot's triad: fever, jaundice, right upper quadrant pain).
- **Gastroenterological Standard of Care & openFDA Pharmacotherapy**: Clinical management of cholestatic liver disorders relies on hydrophilic bile acid replacement using Ursodiol (Actigall, NDA 019594) to displace cytotoxic hydrophobic bile acids, and farnesoid X receptor (FXR) agonists such as Obeticholic acid (Ocaliva, NDA 207999) to downregulate endogenous bile synthesis. Biliary tract bacterial infections necessitate urgent intravenous broad-spectrum coverage with Ampicillin-sulbactam (Unasyn, NDA 050608) or Piperacillin-tazobactam. Low-frequency acoustic waves have no capacity to clear ductal mechanical obstructions or hydrolyze bile stones.
- **Visceral Acoustic Modeling**: The 2.7 Hz frequency ($T \approx 0.370\text{ s}$) sits within the delta spectrum, corresponding to rhythmic mechanotransduction frequencies that interact with smooth muscle visceral pacemakers. In experimental biophysics, 2.7 Hz is studied as a non-invasive sonic resonance to evaluate smooth muscle relaxation and biliary colic-associated muscle guarding.

### 5. Web Audio API Master Synthesizer

```javascript
/**
 * Rife2_7Synthesizer - Web Audio API Synthesis Engine for 2.7 Hz
 * 2.7 Hz is infrasonic, so it is rendered via amplitude modulation of a 200 Hz audible carrier.
 */
class Rife2_7Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.lfo = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Audible carrier wave (200 Hz pure sine)
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(200, t);

    // Infrasonic LFO modulator at 2.7 Hz
    this.lfo = this.audioCtx.createOscillator();
    this.lfo.type = 'sine';
    this.lfo.frequency.setValueAtTime(2.7, t);

    // Modulator gain depth and baseline bias
    const depth = this.audioCtx.createGain();
    depth.gain.setValueAtTime(0.5, t);
    const amp = this.audioCtx.createGain();
    amp.gain.setValueAtTime(0.5, t);

    this.lfo.connect(depth);
    depth.connect(amp.gain);
    this.carrier.connect(amp);
    amp.connect(this.masterGain);

    // Smooth linear attack envelope (3 seconds)
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrier.start(t);
    this.lfo.start(t);
    this.isActive = true;

    // Automatic session shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.lfo.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.lfo) this.lfo.stop();
      this.isActive = false;
    }, 1000);
  }
}
```
