/**
 * Rife875Synthesizer - Web Audio API Synthesis Engine for 875 Hz

class Rife875Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrierOsc = null;
    this.subHarmonic = null;
    this.masterGain = null;
    this.isRunning = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isRunning) return;

    const t = this.audioCtx.currentTime;

    // 875 Hz Fundamental
    this.carrierOsc = this.audioCtx.createOscillator();
    this.carrierOsc.type = 'sine';
    this.carrierOsc.frequency.setValueAtTime(875, t);

    // 437.5 Hz Sub-Harmonic (Cervical lymphatic grounding)
    this.subHarmonic = this.audioCtx.createOscillator();
    this.subHarmonic.type = 'sine';
    this.subHarmonic.frequency.setValueAtTime(437.5, t);

    const subGain = this.audioCtx.createGain();
    subGain.gain.setValueAtTime(0.25, t);

    this.carrierOsc.connect(this.masterGain);
    this.subHarmonic.connect(subGain);
    subGain.connect(this.masterGain);

    // Smooth 3-second attack
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrierOsc.start(t);
    this.subHarmonic.start(t);
    this.isRunning = true;

    // Scheduled shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrierOsc.stop(stopTime);
    this.subHarmonic.stop(stopTime);

    this.carrierOsc.onended = () => {
      this.isRunning = false;
    };
  }

  stop() {
    if (!this.isRunning || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrierOsc) this.carrierOsc.stop();
      if (this.subHarmonic) this.subHarmonic.stop();
      this.isRunning = false;
    }, 1000);
  }
}
