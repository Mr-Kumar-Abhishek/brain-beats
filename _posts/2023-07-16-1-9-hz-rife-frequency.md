### 1. Microbiological & Pathological Profile

- **Degenerative Retinopathy & Bruch's Membrane Pathobiology**: Age-related macular degeneration (AMD) centers on metabolic dysfunction of retinal pigment epithelial (RPE) cells responsible for photoreceptor outer segment phagocytosis. Lipid-rich basomedial deposits (drusen) accumulate between the RPE basement membrane and the inner collagenous layer of Bruch's membrane, impeding nutrient diffusion from the choriocapillaris. In neovascular wet AMD, vascular endothelial growth factor (VEGF) overexpression drives choroidal neovascularization, exudative leakage, and disciform fibrotic scarring.
- **Ophthalmic Pharmacology & FDA Indications**: Definite therapy for neovascular AMD requires intravitreal biologic injections targeting VEGF and Angiopoietin-2, including Aflibercept HD (Eylea HD, BLA 761355 / BLA 125387), Faricimab (Vabysmo, BLA 761235), and Ranibizumab (Lucentis, BLA 125156). Advanced geographic atrophy is addressed with complement C3 inhibitor Pegcetacoplan (Syfovre, NDA 217171) and C5 inhibitor Avacincaptad pegol (Izervay, NDA 217277). Infrasonic frequencies do not neutralize angiogenic factors or dissolve sub-retinal neovascular membranes.

### 5. Web Audio API Master Synthesizer

```javascript
/**
 * Rife1_9Synthesizer - Web Audio API Synthesis Engine for 1.9 Hz
 * 1.9 Hz is infrasonic, so it is rendered via amplitude modulation of a 200 Hz audible carrier.
 */
class Rife1_9Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.lfo = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Audible carrier wave (200 Hz pure sine)
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(200, t);

    // Infrasonic LFO modulator at 1.9 Hz
    this.lfo = this.audioCtx.createOscillator();
    this.lfo.type = 'sine';
    this.lfo.frequency.setValueAtTime(1.9, t);

    // Modulator gain depth and baseline bias
    const depth = this.audioCtx.createGain();
    depth.gain.setValueAtTime(0.5, t);
    const amp = this.audioCtx.createGain();
    amp.gain.setValueAtTime(0.5, t);

    this.lfo.connect(depth);
    depth.connect(amp.gain);
    this.carrier.connect(amp);
    amp.connect(this.masterGain);

    // Smooth linear attack envelope (3 seconds)
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrier.start(t);
    this.lfo.start(t);
    this.isActive = true;

    // Automatic session shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.lfo.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.lfo) this.lfo.stop();
      this.isActive = false;
    }, 1000);
  }
}
```
