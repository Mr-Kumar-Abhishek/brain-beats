### 1. Microbiological & Pathological Profile

- **Multiple Sclerosis (MS) & Neuro-Immune Pathogenesis**: MS is an autoimmune demyelinating disease of the central nervous system characterized by perivascular mononuclear cell infiltration, oligodendrocyte injury, and multifocal axonal transection across the cerebrum, brainstem, and spinal cord. Persistent neuro-inflammatory signaling disrupts central saltatory conduction, presenting clinically with optic neuritis, spasticity, sensory loss, and severe chronic cognitive fatigue.
- **Disease-Modifying Therapies & openFDA Regulatory Data**: High-efficacy disease-modifying therapies (DMTs) are required to arrest active MS progression. The openFDA database identifies pivotal monoclonal antibodies targeting CD20+ B-lymphocytes, including Ocrelizumab (Ocrevus, BLA 761053) and Ofatumumab (Kesimpta, BLA 761170), as well as alpha-4 integrin antagonists like Natalizumab (Tysabri, BLA 125104) that prevent lymphocyte extravasation across the blood-brain barrier. Acoustic vibrations cannot modulate cellular adhesion molecules or stop auto-reactive T-cell clonal expansion.
- **Infrasonic Neuro-Oscillatory Dynamics**: At 2.8 Hz ($T \approx 0.357\text{ s}$), the frequency resides within the slow delta electroencephalographic (EEG) band. In neuro-acoustic research, 2.8 Hz is modeled as a rhythmic entrainment stimulus to downregulate chronic central sympathetic hypertonicity, improve non-REM sleep consolidation, and mitigate secondary neurogenic fatigue in individuals with neuro-inflammatory disorders.

### 5. Web Audio API Master Synthesizer

```javascript
/**
 * Rife2_8Synthesizer - Web Audio API Synthesis Engine for 2.8 Hz
 * 2.8 Hz is infrasonic, so it is rendered via amplitude modulation of a 200 Hz audible carrier.
 */
class Rife2_8Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.lfo = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Audible carrier wave (200 Hz pure sine)
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(200, t);

    // Infrasonic LFO modulator at 2.8 Hz
    this.lfo = this.audioCtx.createOscillator();
    this.lfo.type = 'sine';
    this.lfo.frequency.setValueAtTime(2.8, t);

    // Modulator gain depth and baseline bias
    const depth = this.audioCtx.createGain();
    depth.gain.setValueAtTime(0.5, t);
    const amp = this.audioCtx.createGain();
    amp.gain.setValueAtTime(0.5, t);

    this.lfo.connect(depth);
    depth.connect(amp.gain);
    this.carrier.connect(amp);
    amp.connect(this.masterGain);

    // Smooth linear attack envelope (3 seconds)
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrier.start(t);
    this.lfo.start(t);
    this.isActive = true;

    // Automatic session shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.lfo.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.lfo) this.lfo.stop();
      this.isActive = false;
    }, 1000);
  }
}
```
