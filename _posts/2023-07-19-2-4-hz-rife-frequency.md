### 1. Microbiological & Pathological Profile

- **Cranial Nerve VII Neuropathology & Bell's Palsy**: Bell's palsy is an acute idiopathic peripheral facial nerve (CN VII) paresis typically triggered by herpes simplex virus (HSV-1) reactivation in the geniculate ganglion. Viral inflammation produces severe edema and ischemic compression of the nerve within the narrow, rigid labyrinthine segment of the fallopian canal, culminating in focal conduction block (neuropraxia) or axonal degeneration (axonotmesis) with unilateral facial flaccidity and lagophthalmos.
- **Evidence-Based Pharmacotherapy & FDA Regimens**: The American Academy of Neurology and AAO-HNS guidelines mandate immediate initiation (within 72 hours of onset) of high-dose systemic oral corticosteroids, specifically Prednisone (Deltasone, NDA 017011), to reduce intra-canalicular nerve edema. In severe presentations, adjunctive antiviral therapy with Valacyclovir (Valtrex, NDA 020487) is co-prescribed. Intensive corneal lubrication with FDA-monographed ocular lubricants (Carboxymethylcellulose) is essential to prevent exposure keratitis and permanent corneal ulceration. Sound therapy cannot decompress the bony facial canal or eradicate latent herpesviruses.

### 5. Web Audio API Master Synthesizer

```javascript
/**
 * Rife2_4Synthesizer - Web Audio API Synthesis Engine for 2.4 Hz
 * 2.4 Hz is infrasonic, so it is rendered via amplitude modulation of a 200 Hz audible carrier.
 */
class Rife2_4Synthesizer {
  constructor() {
    this.audioCtx = null;
    this.carrier = null;
    this.lfo = null;
    this.masterGain = null;
    this.isActive = false;
  }

  init() {
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    this.audioCtx = new AudioCtx();

    this.masterGain = this.audioCtx.createGain();
    this.masterGain.gain.setValueAtTime(0.0001, this.audioCtx.currentTime);
    this.masterGain.connect(this.audioCtx.destination);
  }

  start(durationMinutes = 30) {
    if (!this.audioCtx) this.init();
    if (this.isActive) return;

    const t = this.audioCtx.currentTime;

    // Audible carrier wave (200 Hz pure sine)
    this.carrier = this.audioCtx.createOscillator();
    this.carrier.type = 'sine';
    this.carrier.frequency.setValueAtTime(200, t);

    // Infrasonic LFO modulator at 2.4 Hz
    this.lfo = this.audioCtx.createOscillator();
    this.lfo.type = 'sine';
    this.lfo.frequency.setValueAtTime(2.4, t);

    // Modulator gain depth and baseline bias
    const depth = this.audioCtx.createGain();
    depth.gain.setValueAtTime(0.5, t);
    const amp = this.audioCtx.createGain();
    amp.gain.setValueAtTime(0.5, t);

    this.lfo.connect(depth);
    depth.connect(amp.gain);
    this.carrier.connect(amp);
    amp.connect(this.masterGain);

    // Smooth linear attack envelope (3 seconds)
    this.masterGain.gain.exponentialRampToValueAtTime(0.3, t + 3);

    this.carrier.start(t);
    this.lfo.start(t);
    this.isActive = true;

    // Automatic session shutdown
    const stopTime = t + (durationMinutes * 60);
    this.masterGain.gain.setValueAtTime(0.3, stopTime - 3);
    this.masterGain.gain.exponentialRampToValueAtTime(0.0001, stopTime);
    this.carrier.stop(stopTime);
    this.lfo.stop(stopTime);

    this.carrier.onended = () => {
      this.isActive = false;
    };
  }

  stop() {
    if (!this.isActive || !this.audioCtx) return;
    const t = this.audioCtx.currentTime;
    this.masterGain.gain.linearRampToValueAtTime(0.0001, t + 1);
    setTimeout(() => {
      if (this.carrier) this.carrier.stop();
      if (this.lfo) this.lfo.stop();
      this.isActive = false;
    }, 1000);
  }
}
```
